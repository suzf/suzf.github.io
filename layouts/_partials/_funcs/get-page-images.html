{{- /* Overrides Hugo's embedded image lookup used by _internal/opengraph.html and _internal/twitter_cards.html. Posts get their own share card (function/og-card.html); every other page, or a post whose card couldn't be made, falls back to the page's `images` param and then the site-wide `params.images` card, as before. Each entry is the dict shape the embedded templates expect. */ -}} {{- $imgs := slice -}} {{- /* Assign, don't `with partialCached`: the embedded templates call this partial as `with partial …`, which Hugo treats as a decorator, and a nested `with partial… else` breaks its bookkeeping. */ -}} {{- $card := partialCached "function/og-card.html" . .RelPermalink -}} {{- if $card -}} {{- $imgs = $imgs | append (dict "Image" $card "RelPermalink" $card.RelPermalink "Permalink" $card.Permalink) -}} {{- else -}} {{- range .Params.images | default site.Params.images -}} {{- $imgs = $imgs | append (dict "RelPermalink" (. | relURL) "Permalink" (. | absURL)) -}} {{- end -}} {{- end -}} {{- return $imgs -}}
